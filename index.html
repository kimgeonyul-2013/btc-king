const express = require('express'); const fs = require('fs'); const path = require('path'); const app = express(); const DATA_FILE = path.join(__dirname, 'ips.json'); app.set('trust proxy', true); app.use(express.json()); // 데이터 파일 초기화 if (!fs.existsSync(DATA_FILE)) { fs.writeFileSync(DATA_FILE, JSON.stringify([])); } // 1. 프론트엔드 HTML 화면 제공 app.get('/', (req, res) => { res.send(` IP 기록 목록

IP 기록 테스트

버튼을 누르면 현재 접속 IP가 서버에 저장되고 아래 목록에 추가됩니다.

    `); }); // 2. 저장된 IP 목록 조회 API app.get('/api/ips', (req, res) => { const list = JSON.parse(fs.readFileSync(DATA_FILE, 'utf8')); res.json(list); }); // 3. IP 저장 API app.post('/api/ips', (req, res) => { const rawIp = req.headers['x-forwarded-for'] || req.socket.remoteAddress || ''; const clientIp = rawIp.split(',')[0].trim(); const time = new Date().toLocaleString('ko-KR', { timeZone: 'Asia/Seoul' }); const list = JSON.parse(fs.readFileSync(DATA_FILE, 'utf8')); list.unshift({ ip: clientIp, time }); // 최신 기록을 맨 위에 추가 fs.writeFileSync(DATA_FILE, JSON.stringify(list, null, 2)); res.json({ status: 'ok' }); }); app.listen(3000, () => { console.log('실행 중: http://localhost:3000'); });
